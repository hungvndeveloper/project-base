import { Request, Response } from "express";
import * as service from "../services/product.service";

export const getAll = async (_: Request, res: Response) => {
  const data = await service.getAllProducts();
  res.json(data);

export const create = async (req: Request, res: Response) => {
  const { name, price, category } = req.body;
  if (!name || !price || !category) return res.status(400).json({ msg: "Missing fields" });
  if (typeof price !== "number") return res.status(400).json({ msg: "Price must be number" });
  const product = await service.addProduct({ name, price, category });
  res.status(201).json(product);
};

export const update = async (req: Request, res: Response) => {
  const id = req.params.id;
  const updated = await service.updateProduct(id, req.body);
  if (!updated) return res.status(404).json({ msg: "Not found" });
  res.json(updated);
};

export const remove = async (req: Request, res: Response) => {
  const id = req.params.id;
  const deleted = await service.deleteProduct(id);
  if (!deleted) return res.status(404).json({ msg: "Not found" });
  res.json({ msg: "Deleted" });
};

// src/middlewares/auth.middleware.ts
import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";
import dotenv from "dotenv";
dotenv.config();

export const authMiddleware = (req: Request, res: Response, next: NextFunction) => {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return res.status(401).json({ msg: "No token" });
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET!);
    (req as any).user = decoded;
    next();
  } catch {
    res.status(403).json({ msg: "Invalid token" });
  }
};

// src/store.ts
import { configureStore, createSlice, PayloadAction } from "@reduxjs/toolkit";
import { Product } from "./types";

interface ProductState {
  products: Product[];
}

const initialState: ProductState = { products: [] };

const productSlice = createSlice({
  name: "product",
  initialState,
  reducers: {
    setProducts(state, action: PayloadAction<Product[]>) {
      state.products = action.payload;
    },
    updateProduct(state, action: PayloadAction<Product>) {
      const idx = state.products.findIndex(p => p._id === action.payload._id);
      if (idx !== -1) state.products[idx] = action.payload;
    },
    deleteProduct(state, action: PayloadAction<string>) {
      state.products = state.products.filter(p => p._id !== action.payload);
    }
  }
});

export const { setProducts, updateProduct, deleteProduct } = productSlice.actions;

export const store = configureStore({
  reducer: { product: productSlice.reducer }
});

export type RootState = ReturnType<typeof store.getState>;
export type AppDispatch = typeof store.dispatch;

// src/pages/Products.tsx (update with edit/delete + redux)
import { useEffect, useState } from "react";
import axios from "axios";
import { useDispatch, useSelector } from "react-redux";
import { RootState, setProducts, updateProduct, deleteProduct } from "../store";
import { Product } from "../types";

export default function Products({ token }: { token: string }) {
  const dispatch = useDispatch();
  const products = useSelector((state: RootState) => state.product.products);
  const [form, setForm] = useState({ name: "", price: 0, category: "food" });
  const [editId, setEditId] = useState<string | null>(null);

  const fetch = async () => {
    const res = await axios.get("http://localhost:3000/products", { headers: { Authorization: `Bearer ${token}` } });
    dispatch(setProducts(res.data));
  };

  const submit = async () => {
    if (editId) {
      const res = await axios.put(`http://localhost:3000/products/${editId}`, form, {
        headers: { Authorization: `Bearer ${token}` }
      });
      dispatch(updateProduct(res.data));
      setEditId(null);
    } else {
      await axios.post("http://localhost:3000/products", form, { headers: { Authorization: `Bearer ${token}` } });
      fetch();
    }
    setForm({ name: "", price: 0, category: "food" });
  };

  const handleEdit = (p: Product) => {
    setForm({ name: p.name, price: p.price, category: p.category });
    setEditId(p._id);
  };

  const handleDelete = async (id: string) => {
    await axios.delete(`http://localhost:3000/products/${id}`, {
      headers: { Authorization: `Bearer ${token}` }
    });
    dispatch(deleteProduct(id));
  };

  useEffect(() => {
    fetch();
  }, []);

  return (
    <div className="p-6">
      <h1 className="text-2xl mb-4">Products</h1>

      <div className="flex gap-2 mb-4">
        <input className="border p-2" placeholder="Name" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} />
        <input className="border p-2" type="number" placeholder="Price" value={form.price} onChange={(e) => setForm({ ...form, price: Number(e.target.value) })} />
        <select className="border p-2" value={form.category} onChange={(e) => setForm({ ...form, category: e.target.value })}>
          <option value="food">Food</option>
          <option value="clothing">Clothing</option>
          <option value="electronics">Electronics</option>
        </select>
        <button className="bg-green-500 text-white px-4 py-2" onClick={submit}>{editId ? "Update" : "Add"}</button>
      </div>

      <ul>
        {products.map((p) => (
          <li key={p._id} className="mb-2">
            <strong>{p.name}</strong> - ${p.price} [{p.category}]
            <button className="ml-4 text-blue-600" onClick={() => handleEdit(p)}>Edit</button>
            <button className="ml-2 text-red-600" onClick={() => handleDelete(p._id)}>Delete</button>
          </li>
        ))}
      </ul>
    </div>
  );
}

            - Quản lý logic rõ ràng, có thể debug qua Redux Devtools

        ❌ Nhược điểm:
            - Cần setup ban đầu
            - Có thể dư thừa nếu project chỉ có 1-2 component

✍️ Khi nào nên dùng Redux?
    Trường hợp	                                            Dùng Redux?
    App nhỏ, 1-2 component, ít tương tác	                ❌ Không cần
    App có nhiều trang (Products, Detail, Cart...)	        ✅ Cần thiết
    Nhiều thành phần dùng chung dữ liệu (auth, user info)	✅ Rất tốt
    Cần cache, offline, persist dữ liệu	                    ✅ Tốt

    Quản lý dữ liệu	            useState	                        useSelector/useDispatch
    Chia sẻ giữa components	    props hoặc context	                store toàn cục
    Hiệu năng xử lý dữ liệu	    fetch toàn bộ sau mỗi thao tác	    cập nhật 1 phần, tốt hơn
    Mức độ tổ chức	            đơn giản	                        chuyên nghiệp, scale tốt

🆚 So sánh Truyền Props Truyền Thống vs Redux
    Tiêu chí	            Truyền Props	                            Redux
    ⚙️ Cơ chế	            Dữ liệu đi từ cha → con qua props	        Dữ liệu lưu ở store toàn cục, component dùng trực tiếp
    🧠 State ở đâu	        useState (local trong component)	        Redux store (global)
    🔁 Update dữ liệu	    Phải truyền cả hàm cập nhật xuống	        Dispatch action từ bất kỳ component nào
    🔄 Dữ liệu nhiều tầng	Truyền props xuyên tầng (prop drilling)	    Dùng useSelector ở bất cứ đâu
    🧹 Tổ chức code	        Dễ rối nếu nhiều cấp	                    Gọn, rõ ràng hơn
    ✅ Phù hợp	           App nhỏ	                                   App nhiều component, nhiều page

🎯 Ví dụ minh hoạ
    📌 Tình huống:
        - Bạn có App → ProductPage → ProductList → ProductItem
          Và bạn cần sửa sản phẩm từ ProductItem.

    🔴 Cách 1: Truyền props truyền thống
        // App.tsx
        const [products, setProducts] = useState([...]);

        <ProductPage products={products} setProducts={setProducts} />

        // ProductPage.tsx
        <ProductList products={products} setProducts={setProducts} />

        // ProductList.tsx
        {products.map(p => (
            <ProductItem product={p} setProducts={setProducts} />
        ))}

        // ProductItem.tsx
        <button onClick={() => {
            const newProducts = [...]; // clone và update
            setProducts(newProducts);
        }}>Edit</button>

        ⛔ Vấn đề:
            - setProducts phải truyền qua 3 tầng → dễ sai, khó bảo trì
            - Gặp prop drilling (chuyền props nhiều tầng)
            - Mỗi component đều phụ thuộc vào cha

    🟢 Cách 2: Dùng Redux
        // ProductItem.tsx
        const dispatch = useDispatch();

        <button onClick={() => dispatch(updateProduct(product))}>Edit</button>
        
        - Không cần nhận setProducts từ props
        - Dùng useSelector(state => state.product.products) ở bất cứ đâu

        ✅ Lợi ích:
            - Dễ chia component nhỏ mà không lo truyền props
            - Tăng khả năng tái sử dụng và test dễ hơn
            - Code ngắn hơn, tổ chức gọn gàng hơn

💡 Tổng kết bảng so sánh
    So sánh	                    Props Truyền Thống	                Redux
    Đơn giản ban đầu	        ✅	                              ❌ (phải setup)
    Dễ scale lên nhiều tầng	    ❌	                              ✅
    Reuse component	            Khó nếu phụ thuộc state cha	        Dễ dàng
    Logic cập nhật phức tạp	    Khó tổ chức	                        Dễ chia nhỏ reducer
    Quản lý toàn app	        Không khả thi	                    Có (auth, cart, etc)

✅ Kết luận
    - App nhỏ: Props là đủ
    - App lớn, đa trang, nhiều logic (auth, cart, product, user...): Dùng Redux là chuẩn bài
